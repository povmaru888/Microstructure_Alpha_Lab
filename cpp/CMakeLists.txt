cmake_minimum_required(VERSION 3.16)
project(microalpha_features VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(microalpha_core src/features.cpp src/csv_io.cpp src/cli.cpp)
target_include_directories(microalpha_core PUBLIC include)
if(MSVC)
  target_compile_options(microalpha_core PRIVATE /W4)
else()
  target_compile_options(microalpha_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(microalpha_features src/main.cpp)
target_link_libraries(microalpha_features PRIVATE microalpha_core)

enable_testing()
add_executable(test_build tests/test_build.cpp)
target_link_libraries(test_build PRIVATE microalpha_core)
add_test(NAME build_smoke COMMAND test_build)
add_executable(test_csv_io tests/test_csv_io.cpp)
target_link_libraries(test_csv_io PRIVATE microalpha_core)
add_test(NAME csv_io COMMAND test_csv_io)
add_executable(test_features_l1 tests/test_features_l1.cpp)
target_link_libraries(test_features_l1 PRIVATE microalpha_core)
add_test(NAME features_l1 COMMAND test_features_l1)
add_executable(test_features_ofi tests/test_features_ofi.cpp)
target_link_libraries(test_features_ofi PRIVATE microalpha_core)
add_test(NAME features_ofi COMMAND test_features_ofi)
add_executable(test_features_depth tests/test_features_depth.cpp)
target_link_libraries(test_features_depth PRIVATE microalpha_core)
add_test(NAME features_depth COMMAND test_features_depth)
add_executable(test_cli tests/test_cli.cpp)
target_link_libraries(test_cli PRIVATE microalpha_core)
add_test(NAME cli COMMAND test_cli)
